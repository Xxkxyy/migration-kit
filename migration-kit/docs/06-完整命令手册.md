## 1. 现场勘查

```bash
# 1.1 端口连通（不依赖 nc / telnet）
for ip in 10.0.0.221 10.0.0.222 10.1.0.196 10.1.0.197; do
  timeout 3 bash -c "echo > /dev/tcp/$ip/22" && echo "$ip SSH OPEN" || echo "$ip SSH CLOSED"
done

# 1.2 是否同一广播域（决定能不能换 IP）★
ip addr add 10.0.0.250/24 dev ens7
ping -c2 -W2 -I 10.0.0.250 10.0.0.254
ip neigh show 10.0.0.254            # REACHABLE=同二层；INCOMPLETE/FAILED=不同 VLAN
ip addr del 10.0.0.250/24 dev ens7

# 1.3 两侧网关是否同一台设备（比 MAC）
ip neigh | grep '\.254 '

# 1.4 跨网段路由是否通
ping -c2 10.1.0.196
traceroute -n -m 3 10.1.0.196

# 1.5 批量取每台的身份（一条命令证明"我是谁"）
for ip in 10.0.0.221 10.1.0.196; do
  echo -n "$ip: "
  ssh -o StrictHostKeyChecking=no -o ConnectTimeout=5 root@$ip 'hostname; ip -o -4 addr show scope global | awk "{print \$4}" | head -1' 2>/dev/null | tr '\n' ' '
  echo
done
```

## 2. 系统与磁盘

```bash
# 2.1 系统画像
cat /etc/os-release | head -3
uname -r; nproc; free -g
date '+%Z %z'; ls -l /etc/localtime
getenforce
bash -c 'ulimit -n; ulimit -u'
cat /proc/sys/vm/swappiness

# 2.2 磁盘总览
df -hT | grep -vE 'tmpfs|overlay|devtmpfs'
lsblk -o NAME,SIZE,TYPE,FSTYPE,MOUNTPOINT
blkid
findmnt -no SOURCE,TARGET,FSTYPE,OPTIONS /data
df -i | grep -vE 'tmpfs|overlay'

# 2.3 谁占了空间
du -sh /data/*/ | sort -h | tail -15
du -sh /data/mysql/data/*/ | sort -h | tail -15

# 2.4 4K 同步写 IOPS（迁移第一指标）
dd if=/dev/zero of=/tmp/io bs=4k count=500 oflag=direct,dsync
rm -f /tmp/io

# 2.5 顺序读写
dd if=/dev/zero of=/tmp/seq bs=1M count=1024 oflag=direct
dd if=/tmp/seq of=/dev/null bs=1M count=1024 iflag=direct
rm -f /tmp/seq

# 2.6 IO 调度器与队列
cat /sys/block/sdb/queue/scheduler
cat /sys/block/sdb/queue/nr_requests

# 2.7 LVM 合并多盘（本次把 1T+1T 合成 2T）
bash scripts/lib/lvm_merge.sh /dev/vdb /dev/vdc            # 预演
bash scripts/lib/lvm_merge.sh /dev/vdb /dev/vdc --yes      # 执行

# 2.8 扩容已有 LV
lvextend -l +100%FREE /dev/vg_data/lv_data
xfs_growfs /data

# 2.9 写入 fstab（用 UUID，别用 /dev/sdb）
UUID=$(blkid -s UUID -o value /dev/vg_data/lv_data)
echo "UUID=$UUID /data xfs defaults 0 0" >> /etc/fstab
findmnt --verify
```

```bash

LimitNOFILE=65536
[Install]

# 6.6 设 root 口令
/usr/local/mysql/bin/mysql -uroot --skip-password -e \
  "alter user 'root'@'localhost' identified by 'xxx'; flush privileges;"
```

### 12.5 外部 Oracle 数据源（动态数据源）

```bash

  where table_schema='ek_manager' and (column_name like '%url%' or column_name like '%driver%')"

  timeout 4 bash -c "echo > /dev/tcp/$h/1521" && echo "$h OPEN" || echo "$h CLOSED"
done

# 4) 服务器没外网时，本地下载再上传
#    本地(Windows): Invoke-WebRequest https://repo1.maven.org/maven2/com/oracle/database/jdbc/ojdbc8/19.3.0.0/ojdbc8-19.3.0.0.jar

# 5) 确认 jar 布局（Spring Boot 2.x = BOOT-INF/lib）
```

## 20. 禁止清单（本次全部踩过，逐条都是事故）

| 禁止 | 原因 | 正确做法 |
|---|---|---|
| `ps -ef \| grep '[_x]' \| xargs kill -9` | 方括号是字符类，杀全机进程（曾杀 systemd/sshd/mysqld） | `pkill -x` 或打印 PID 人工确认 |
| `rsync --delete` | 源端日志轮转会删目标端文件 | 只补不删 |
| `DELETE FROM 大表 WHERE id BETWEEN ...` | 慢盘逐行删，26 分钟零吞吐 + 回滚 40 分钟 | `TRUNCATE` + 重新分段导入 |
| `INSERT IGNORE` 反复重试同一段 | 行数叠加，表膨胀 11~14 倍 | 先清空，或 `--insert-ignore` 单次 |
| 分段边界从目标端取 | TRUNCATE 后 MIN/MAX 为 NULL，RC=0 但导 0 行 | 边界必须从**源端**取 |
| 不做 `ANALYZE` 就切生产 | 优化器认为表空，查询 90 秒起、网关 504 | 切参数时同步 `stats_auto_recalc=ON` + 全库 ANALYZE |
| 把 `Entrypoint` 和 `Cmd` 一起拼 | `weed weed server` / `java java -jar` 启动即退出 | 镜像有 ENTRYPOINT 时只传 Cmd |
| 只改配置文件就 `docker restart` | Env/Cmd 里的地址不读文件 | 重建容器（`--force`） |
| 用 `--exclude-dir=data` 统计命中 | 整个业务目录被排除，结果永远 0 | 精确排除或改用 include |
| SQL 里直接写反引号 | shell 命令替换，库名变空 | SQL 写文件再执行 |
| 在 PowerShell 里内联长命令 | `2>/dev/null`、`$()`、`$_` 被解析，PSReadLine 崩 | 写成 .py/.sh 文件再执行 |
| 并行跑两个同名长任务 | 互相 TRUNCATE 同一张表 | 入口 `flock -n` |
| 用 `recv_exit_status()` 不设超时 | TCP 半开时永久阻塞，线程静默死亡 | 旁路线程 + join 超时 + 丢弃连接 |
